def init_db() -> None:
    try:
        with sqlite3.connect('bot_database.db', timeout=15.0) as conn:
            cursor = conn.cursor()
            cursor.execute('''CREATE TABLE IF NOT EXISTS bots (id INTEGER PRIMARY KEY AUTOINCREMENT, token TEXT UNIQUE, username TEXT, owner_id INTEGER)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS blacklist (user_id INTEGER PRIMARY KEY)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS hidden_files (id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, filename TEXT, content TEXT)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS admin_auth (user_id INTEGER PRIMARY KEY, is_logged_in INTEGER DEFAULT 0)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS granted_admins (user_id INTEGER PRIMARY KEY)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS granted_vip (user_id INTEGER PRIMARY KEY)''')
            cursor.execute('''CREATE TABLE IF NOT EXISTS audit_logs (id INTEGER PRIMARY KEY AUTOINCREMENT, timestamp REAL, user_id INTEGER, action TEXT, details TEXT)''')
            conn.commit()
    except Exception as e:
        print(f"Lỗi khởi tạo DB: {e}")

init_db()

def get_secret_password() -> str:
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT value FROM settings WHERE key = 'admin_secret_password'")
            row = cursor.fetchone()
            if row: return row[0]
    except Exception: pass
    return DEFAULT_SECRET_PASSWORD

def set_secret_password(new_pw: str) -> None:
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("INSERT OR REPLACE INTO settings (key, value) VALUES ('admin_secret_password', ?)", (new_pw,))
            conn.commit()
    except Exception: pass

def log_audit(user_id: int, action: str, details: str) -> None:
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("INSERT INTO audit_logs (timestamp, user_id, action, details) VALUES (?, ?, ?, ?)", (time.time(), user_id, action, details))
            conn.commit()
    except Exception: pass

def is_admin(user_id: int) -> bool:
    if user_id == ADMIN_ID: return True
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT 1 FROM granted_admins WHERE user_id = ?", (user_id,))
            if cursor.fetchone(): return True
    except Exception: pass
    return False

def is_admin_logged_in(user_id: int) -> bool:
    if user_id == ADMIN_ID: return True
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT is_logged_in FROM admin_auth WHERE user_id = ?", (user_id,))
            row = cursor.fetchone()
            if row and row[0] == 1: return True
    except Exception: pass
    return False

def set_admin_logged_in(user_id: int, status: int) -> None:
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("INSERT OR REPLACE INTO admin_auth (user_id, is_logged_in) VALUES (?, ?)", (user_id, status))
            conn.commit()
    except Exception: pass

def is_vip(user_id: int) -> bool:

            cursor.execute("SELECT 1 FROM granted_vip WHERE user_id = ?", (user_id,))

def is_authorized(user_id: int) -> bool:

@client.on(events.CallbackQuery(pattern=r'func_stopbot_(\d+)'))
async def func_stopbot_cb(event) -> None:
    chat_id = event.chat_id
    if chat_id in ACTIVE_TASKS:
        for mode in list(ACTIVE_TASKS[chat_id].keys()):
            for task in ACTIVE_TASKS[chat_id][mode]: task.cancel()
        del ACTIVE_TASKS[chat_id]
    gc.collect()
    await event.answer("⚡ Đã thực hiện StopBot tổng dọn dẹp RAM thành công!", alert=True)

@client.on(events.CallbackQuery(pattern=r'view_audit_log_(\d+)'))
async def view_audit_log_cb(event) -> None:
    try:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT timestamp, user_id, action, details FROM audit_logs ORDER BY id DESC LIMIT 5")
            rows = cursor.fetchall()
        if not rows:
            await event.answer("Nhật ký trống!", alert=True)
            return
        log_text = "📋 **NHẬT KÝ HOẠT ĐỘNG GẦN NHẤT:**\n"
        for ts, uid, act, det in rows:
            t_str = time.strftime('%H:%M:%S', time.localtime(ts))
            log_text += f"• `[{t_str}]` User `{uid}` | **{act}**: {det}\n"
        await event.respond(log_text, parse_mode='markdown')
    except Exception as e:
        await event.answer(f"Lỗi: {e}", alert=True)

@client.on(events.NewMessage(pattern=r'\.capquyen'))
async def cap_quyen_command(event) -> None:
    user_id = event.sender_id
    if not is_admin(user_id) or not is_admin_logged_in(user_id):
        await event.respond("⚠️ Bạn không có quyền sử dụng lệnh này!", parse_mode='markdown')
        return

    target_user_id = None
    target_name = "Không rõ"
    if event.is_reply:
        reply_msg = await event.get_reply_message()
        if reply_msg and reply_msg.sender_id:
            target_user_id = reply_msg.sender_id
            if reply_msg.sender:
                target_name = f"@{reply_msg.sender.username}" if reply_msg.sender.username else reply_msg.sender.first_name
    else:
        args = event.raw_text.split()
        if len(args) < 2:
            await event.respond("⚠️ Sai cú pháp! Dùng: `.capquyen <user_id>` hoặc reply tin nhắn.", parse_mode='markdown')
            return
        target_user_id = int(args[1].strip())
        target_name = f"ID: {target_user_id}"

    if target_user_id:
        with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
            cursor = conn.cursor()
            cursor.execute("INSERT OR IGNORE INTO granted_vip (user_id) VALUES (?)", (target_user_id,))
            cursor.execute("INSERT OR IGNORE INTO granted_admins (user_id) VALUES (?)", (target_user_id,))
            conn.commit()
        log_audit(user_id, "GRANT_ACCESS", f"Cấp quyền cho user {target_user_id}")
        await event.respond(f"✅ **Đã cấp quyền VIP và Admin thành công cho:** `{target_name}` (`{target_user_id}`)", parse_mode='markdown')

@client.on(events.NewMessage(pattern=r'\.addbot'))
async def add_bot_command(event) -> None:
    user_id = event.sender_id
    if not is_admin(user_id) or not is_admin_logged_in(user_id): return
    args = event.raw_text.split(maxsplit=2)
    if len(args) < 3: return
    token, username = args[1], args[2].lstrip('@')
    with sqlite3.connect('bot_database.db', timeout=5.0) as conn:
        cursor = conn.cursor()
        cursor.execute("INSERT OR REPLACE INTO bots (token, username, owner_id) VALUES (?, ?, ?)", (token, username, user_id))
        conn.commit()
    await event.respond(f"✅ **Đã thêm bot phụ:** `@{username}`", parse_mode='markdown')

@client.on(events.NewMessage(func=lambda e: not e.text or not e.text.startswith('.')))
async def handle_incoming_messages(event) -> None:
    user_id = event.sender_id
    if not is_authorized(user_id) or event.is_group: return
    
    if user_id in USER_SESSIONS and USER_SESSIONS[user_id].get("step") == "waiting_for_new_password":
        new_pw = event.raw_text.strip()
        if len(new_pw) < 4:
            await event.respond("⚠️ Mật khẩu quá ngắn! Nhập lại:", parse_mode='markdown')
            return
        set_secret_password(new_pw)
        del USER_SESSIONS[user_id]
        log_audit(user_id, "CHANGE_ADMIN_PASSWORD", "Đổi mật khẩu ẩn thành công.")
        await event.respond(f"✅ **Đổi mật khẩu đăng nhập ẩn thành công!** Mật khẩu mới: `{new_pw}`", parse_mode='markdown')
        raise events.StopPropagation
