@bot.message_handler(commands=['upfile'])
def upload_permanent_file(message):
    if message.from_user.id != ADMIN_ID:
        bot.reply_to(message, "❌ Chỉ có Admin tối cao mới có quyền thêm file mới vào hệ thống!")
        return
        
    if not message.document and not message.reply_to_message:
        bot.reply_to(message, "⚠️ Vui lòng gửi file hoặc reply file kèm lệnh `/upfile`!", parse_mode="Markdown")
        return

    target_doc = message.document if message.document else message.reply_to_message.document
    try:
        file_info = bot.get_file(target_doc.file_id)
        file_name = target_doc.file_name
        downloaded_file = bot.download_file(file_info.file_path)
        
        file_path = os.path.join(UPLOAD_FOLDER, file_name)
        with open(file_path, 'wb') as f:
            f.write(downloaded_file)
            
        bot.reply_to(message, f"✅ Đã lưu file `{file_name}` vào hệ thống thành công!", parse_mode="Markdown")
    except Exception as e:
        bot.reply_to(message, f"❌ Lỗi khi lưu file: {str(e)}")

# --- LỆNH /TOXIC: ĐỌC TỪNG DÒNG TRONG FILE + VÒNG LẶP + @ USER ---
def run_toxic_loop(chat_id, file_path, delay, mention_target):
    try:
        while True:
            if not os.path.exists(file_path):
                break
            with open(file_path, 'r', encoding='utf-8') as f:
                lines = f.readlines()
                
            if not lines:
                break
                
            for line in lines:
                content = line.strip()
                if not content:
                    continue
                
                if mention_target:
                    final_message = f"{mention_target} {content}"
                else:
                    final_message = content
                    
                bot.send_message(chat_id, final_message)
                time.sleep(float(delay))
    except Exception as e:
        print(f"Lỗi tiến trình toxic: {e}")

@bot.message_handler(commands=['toxic'])
def cmd_toxic(message):
    user_id = message.from_user.id
    if user_id != ADMIN_ID:
        bot.reply_to(message, "❌ Chức năng này chỉ dành riêng cho Admin!")
        return

    parts = message.text.split(maxsplit=3)
    files = os.listdir(UPLOAD_FOLDER)
    
    if len(parts) < 3:
        file_list_str = "\n".join([f"📁 `{f}`" for f in files]) if files else "*(Trống)*"
        bot.reply_to(
            message,
            f"⚠️ **Cú pháp lệnh /toxic chưa đúng!**\n"
            f"👉 Dùng: `/toxic [delay] [tên_file] [@username]`\n\n"
            f"📂 **Các file có sẵn:**\n{file_list_str}",
            parse_mode="Markdown"
        )
        return

    try:
        delay_time = float(parts[1])
        file_name = parts[2]
        mention_target = parts[3] if len(parts) > 3 else ""
        
        target_path = os.path.join(UPLOAD_FOLDER, file_name)
        if not os.path.exists(target_path):
            bot.reply_to(message, f"❌ Không tìm thấy file `{file_name}` trong hệ thống!")
            return

        bot.reply_to(
            message,
            f"🔥 **Đã kích hoạt TOXIC (Từng dòng + Vòng lặp)!**\n"
            f"⏱️ Delay: `{delay_time}` giây/dòng\n"
            f"📁 File: `{file_name}`\n"
            f"🎯 Mục tiêu: `{mention_target if mention_target else 'Không có'}`",
            parse_mode="Markdown"
        )

        task_thread = threading.Thread(target=run_toxic_loop, args=(message.chat.id, target_path, delay_time, mention_target))
        task_thread.daemon = True
        task_thread.start()

    except Exception as e:
        bot.reply_to(message, f"❌ Có lỗi xảy ra: {str(e)}")

# --- CÁC LỆNH KHÁC (/spam, /treongon) ---
def run_full_file_task(chat_id, file_path, delay, cmd_name):
    try:
        if not os.path.exists(file_path):
            return
        with open(file_path, 'r', encoding='utf-8') as f:
            file_content = f.read()
            
        while True:
            bot.send_message(chat_id, f"📁 [{cmd_name.upper()}]\n{file_content}")
            time.sleep(float(delay))
    except Exception as e:
        print(f"Lỗi tiến trình {cmd_name}: {e}")
